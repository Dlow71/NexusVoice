## 1️⃣ StructuralMarkdownProcessor

### 职责

- 解析Markdown为AST树
- 按语义结构分段
- 输出ProcessedSegment列表

### 核心代码

```java
@Component("structuralMarkdownProcessor")
public class StructuralMarkdownProcessor implements MarkdownProcessor {
    
    private final MarkdownAstParser astParser;
    private final DocumentTreeBuilder treeBuilder;
    private MarkdownProcessorProperties properties;
    private boolean rawMode = false;  // 原文模式标志
    
    @Override
    public List<ProcessedSegment> processToSegments(
        String markdown, 
        ProcessingContext context
    ) {
        if (markdown == null || markdown.trim().isEmpty()) {
            return new ArrayList<>();
        }
        
        try {
            if (rawMode) {
                return processRawSegments(markdown, context);
            } else {
                return processRegularSegments(markdown, context);
            }
        } catch (Exception e) {
            log.error("处理失败", e);
            // 回退：整个文档作为一个段落
            return List.of(new ProcessedSegment(markdown, SegmentType.TEXT, null));
        }
    }
    
    // 原文模式：保持原始内容
    private List<ProcessedSegment> processRawSegments(
        String markdown, 
        ProcessingContext context
    ) {
        Node document = astParser.parse(markdown);
        DocumentTree tree = treeBuilder.buildRawDocumentTree(
            document, 
            properties.getSegmentSplit()
        );
        List<ProcessedSegment> segments = tree.performHierarchicalSplit();
        
        // 设置顺序
        for (int i = 0; i < segments.size(); i++) {
            segments.get(i).setOrder(i);
        }
        
        return segments;
    }
    
    // 设置原文模式
    public void setRawMode(boolean rawMode) {
        this.rawMode = rawMode;
    }
}
```

## 2️⃣ DocumentTreeBuilder

### 职责

- 将AST节点转换为DocumentTree
- 支持原文模式和翻译模式
- 构建节点层次关系

### 核心代码

```java
@Component
public class DocumentTreeBuilder {
    
    private final MarkdownTextExtractor textExtractor;
    
    public DocumentTree buildRawDocumentTree(
        Node document, 
        SegmentSplitConfig config
    ) {
        DocumentTree tree = new DocumentTree(config);
        
        for (Node child : document.getChildren()) {
            processNode(child, tree);
        }
        
        return tree;
    }
    
    private void processNode(Node node, DocumentTree tree) {
        if (node instanceof Heading heading) {
            int level = heading.getLevel();
            String content = textExtractor.extractText(heading);
            tree.addHeading(level, content, heading);
            
        } else if (node instanceof Paragraph paragraph) {
            String content = textExtractor.extractText(paragraph);
            tree.addParagraph(content, paragraph);
            
        } else if (node instanceof FencedCodeBlock codeBlock) {
            String code = codeBlock.getContentChars().toString();
            String lang = codeBlock.getInfo().toString();
            tree.addCodeBlock(code, lang, codeBlock);
            
        } else if (node instanceof TableBlock table) {
            String tableContent = extractTable(table);
            tree.addTable(tableContent, table);
            
        } else if (node instanceof BulletList list) {
            String listContent = textExtractor.extractText(list);
            tree.addList(listContent, list);
        }
    }
}
```

### DocumentTree类

```java
public class DocumentTree {
    
    private final List<DocumentNode> nodes = new ArrayList<>();
    private final SegmentSplitConfig config;
    
    public List<ProcessedSegment> performHierarchicalSplit() {
        List<ProcessedSegment> segments = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        int currentLength = 0;
        
        for (DocumentNode node : nodes) {
            String content = node.getContent();
            int length = content.length();
            
            // 一级标题强制分段
            if (node.isHeading() && node.getLevel() == 1) {
                if (currentLength > 0) {
                    segments.add(createSegment(current.toString()));
                    current.setLength(0);
                    currentLength = 0;
                }
            }
            
            // 长度超限分段
            if (currentLength + length > config.getMaxLength() 
                && currentLength > config.getMinLength()) {
                segments.add(createSegment(current.toString()));
                current.setLength(0);
                currentLength = 0;
            }
            
            current.append(content).append("\n\n");
            currentLength += length;
        }
        
        if (currentLength > 0) {
            segments.add(createSegment(current.toString()));
        }
        
        return segments;
    }
}
```

## 3️⃣ MarkdownRagDocumentProcessing

### 职责

- Markdown文档处理策略
- 实现两阶段处理架构
- 协调各组件完成处理

### 核心代码

```java
@Service("markdown")
public class MarkdownRagDocumentProcessing 
    extends AbstractDocumentProcessingStrategy {
    
    private final StructuralMarkdownProcessor structuralProcessor;
    private final DocumentVectorizationOrchestrator vectorOrchestrator;
    private final DocumentUnitRepository documentUnitRepository;
    private final UserModelConfigResolver modelConfigResolver;
    
    @Override
    public void handle(RagDocMessage message, String strategy) throws Exception {
        log.info("开始处理Markdown文档: {}", message.getFileId());
        super.handle(message, strategy);  // 调用模板方法
    }
    
    @Override
    public Map<Integer, String> processFile(
        byte[] fileBytes, 
        int totalPages, 
        RagDocMessage message
    ) {
        String markdown = new String(fileBytes, StandardCharsets.UTF_8);
        
        // 构建处理上下文
        ProcessingContext context = ProcessingContext.from(
            message, 
            modelConfigResolver
        );
        
        // 阶段1：原文拆分
        structuralProcessor.setRawMode(true);
        List<ProcessedSegment> segments = structuralProcessor.processToSegments(
            markdown, 
            context
        );
        
        // 转换为Map返回
        Map<Integer, String> ocrData = new HashMap<>();
        for (int i = 0; i < segments.size(); i++) {
            ocrData.put(i, segments.get(i).getContent());
        }
        
        return ocrData;
    }
    
    @Override
    public void insertData(
        RagDocMessage message, 
        Map<Integer, String> ocrData
    ) throws Exception {
        List<DocumentUnitEntity> savedUnits = new ArrayList<>();
        
        // 保存原文段落
        for (int i = 0; i < ocrData.size(); i++) {
            DocumentUnitEntity unit = new DocumentUnitEntity();
            unit.setContent(ocrData.get(i));
            unit.setPage(i);
            unit.setFileId(message.getFileId());
            unit.setIsOcr(true);
            unit.setIsVector(false);
            
            documentUnitRepository.insert(unit);
            savedUnits.add(unit);
        }
        
        log.info("阶段1完成: 保存{}个段落", savedUnits.size());
        
        // 阶段2：触发向量化
        ProcessingContext context = ProcessingContext.from(
            message, 
            modelConfigResolver
        );
        vectorOrchestrator.processDocumentUnits(savedUnits, context);
    }
}
```

## 4️⃣ DocumentVectorizationOrchestrator

### 职责

- 编排翻译、分割、向量化流程
- 处理DocumentUnit列表
- 触发向量化MQ消息

### 核心代码

```java
@Service
public class DocumentVectorizationOrchestrator {
    
    private final MarkdownAstRewriter translator;
    private final MarkdownContentSplitter splitter;
    private final DocumentUnitRepository repository;
    private final MessagePublisher messagePublisher;
    
    public void processDocumentUnits(
        List<DocumentUnitEntity> units, 
        ProcessingContext context
    ) {
        for (DocumentUnitEntity unit : units) {
            try {
                processSingleUnit(unit, context);
            } catch (Exception e) {
                log.error("处理失败: {}", unit.getId(), e);
            }
        }
    }
    
    public void processSingleUnit(
        DocumentUnitEntity unit, 
        ProcessingContext context
    ) {
        String originalContent = unit.getContent();
        
        // 步骤1：翻译特殊节点
        String translated = translator.translateSpecialNodes(
            originalContent, 
            context
        );
        
        // 步骤2：智能二次分割
        String titleContext = extractTitleContext(unit);
        List<String> vectorTexts = splitter.splitIfNeeded(
            translated, 
            titleContext
        );
        
        log.info("文档单元{}分割为{}个片段", unit.getId(), vectorTexts.size());
        
        // 步骤3：触发向量化
        for (int i = 0; i < vectorTexts.size(); i++) {
            triggerVectorization(unit, vectorTexts.get(i), i, context);
        }
        
        // 步骤4：更新状态
        unit.setIsVector(true);
        repository.updateById(unit);
    }
    
    private void triggerVectorization(
        DocumentUnitEntity unit,
        String vectorText,
        int segmentIndex,
        ProcessingContext context
    ) {
        RagDocSyncStorageMessage message = new RagDocSyncStorageMessage();
        message.setId(unit.getId() + "_segment_" + segmentIndex);
        message.setFileId(unit.getFileId());
        message.setPage(unit.getPage() * 1000 + segmentIndex);
        message.setContent(vectorText);
        message.setUserId(context.getUserId());
        message.setEmbeddingModelConfig(context.getEmbeddingModelConfig());
        
        // 发送MQ消息
        MessageEnvelope<RagDocSyncStorageMessage> envelope = 
            MessageEnvelope.builder(message)
                .addEventType(EventType.DOC_SYNC_RAG)
                .build();
        messagePublisher.publish(RagDocSyncStorageEvent.route(), envelope);
    }
}
```

## 5️⃣ MarkdownAstRewriter

### 职责

- 翻译特殊节点为自然语言
- 基于AST重写文档
- 调用LLM进行翻译

### 核心代码

```java
@Component
public class MarkdownAstRewriter {
    
    private final NodeTranslatorService translatorService;
    private final Parser parser;
    
    public String translateSpecialNodes(
        String originalContent, 
        ProcessingContext context
    ) {
        if (originalContent == null || originalContent.trim().isEmpty()) {
            return originalContent;
        }
        
        try {
            // 解析为AST
            Node document = parser.parse(originalContent);
            
            // 重写文档
            StringBuilder result = new StringBuilder();
            rewriteDocument(document, result, context);
            
            return result.toString();
        } catch (Exception e) {
            log.error("翻译失败: {}", e.getMessage(), e);
            return originalContent;  // 降级：返回原文
        }
    }
    
    private void rewriteDocument(
        Node document, 
        StringBuilder result, 
        ProcessingContext context
    ) {
        for (Node child : document.getChildren()) {
            rewriteNode(child, result, context);
        }
    }
    
    private void rewriteNode(
        Node node, 
        StringBuilder result, 
        ProcessingContext context
    ) {
        if (node instanceof FencedCodeBlock codeBlock) {
            // 翻译代码块
            String code = codeBlock.getContentChars().toString();
            String lang = codeBlock.getInfo().toString();
            String translated = translatorService.translateCodeBlock(
                code, lang, context
            );
            result.append(translated).append("\n\n");
            
        } else if (node instanceof TableBlock table) {
            // 翻译表格
            String tableMarkdown = extractTable(table);
            String translated = translatorService.translateTable(
                tableMarkdown, context
            );
            result.append(translated).append("\n\n");
            
        } else if (node instanceof Image image) {
            // OCR图片
            String url = image.getUrl().toString();
            String translated = translatorService.ocrImage(url, context);
            result.append(translated).append("\n\n");
            
        } else {
            // 普通节点保持不变
            result.append(node.getChars()).append("\n");
        }
    }
}
```

## 6️⃣ MarkdownContentSplitter

### 职责

- 智能二次分割
- 三层分割策略
- 重叠机制

### 核心代码

```java
@Component
public class MarkdownContentSplitter {
    
    @Value("${rag.vector.max-length:1800}")
    private int maxVectorLength;
    
    @Value("${rag.vector.overlap-size:100}")
    private int overlapSize;
    
    public List<String> splitIfNeeded(String content, String titleContext) {
        String fullContent = buildContentWithContext(content, titleContext);
        
        if (fullContent.length() <= maxVectorLength) {
            return List.of(fullContent);
        }
        
        // 三层策略
        List<String> chunks;
        if (isValid(chunks = splitByParagraphs(fullContent, titleContext))) {
            return chunks;
        }
        if (isValid(chunks = splitBySentences(fullContent, titleContext))) {
            return chunks;
        }
        return forceSplit(fullContent, titleContext);
    }
    
    private List<String> splitByParagraphs(
        String content, 
        String titleContext
    ) {
        String[] paragraphs = content.split("\n\n");
        List<String> chunks = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        
        String titlePrefix = buildTitlePrefix(titleContext);
        int available = maxVectorLength - titlePrefix.length() - overlapSize;
        
        for (String para : paragraphs) {
            if (para.trim().isEmpty()) continue;
            
            if (current.length() + para.length() > available) {
                if (current.length() > 0) {
                    chunks.add(titlePrefix + current.toString());
                    
                    // 重叠
                    String overlap = extractOverlap(current.toString());
                    current.setLength(0);
                    if (!overlap.isEmpty()) {
                        current.append(overlap).append("\n\n");
                    }
                }
            }
            
            current.append(para).append("\n\n");
        }
        
        if (current.length() > 0) {
            chunks.add(titlePrefix + current.toString());
        }
        
        return chunks;
    }
    
    private String extractOverlap(String text) {
        if (text.length() <= overlapSize) {
            return text;
        }
        return text.substring(text.length() - overlapSize);
    }
}
```

@Component
public class EmbeddingDomainService {
    
    private final EmbeddingModelFactory modelFactory;
    private final EmbeddingStore<TextSegment> embeddingStore;
    
    public void syncStorage(RagDocSyncStorageMessage message) {
        try {
            // 创建embedding模型
            OpenAiEmbeddingModel model = modelFactory.createEmbeddingModel(
                message.getEmbeddingModelConfig()
            );
            
            // 生成向量
            Embedding embedding = model.embed(message.getContent()).content();
            
            // 构建metadata
            Metadata metadata = new Metadata();
            metadata.put("file_id", message.getFileId());
            metadata.put("dataset_id", message.getDatasetId());
            metadata.put("page", message.getPage());
            metadata.put("user_id", message.getUserId());
            
            // 存储
            TextSegment segment = TextSegment.from(message.getContent(), metadata);
            embeddingStore.add(embedding, segment);
            
            log.info("向量化完成: fileId={}, page={}", 
                message.getFileId(), message.getPage());
                
        } catch (Exception e) {
            log.error("向量化失败", e);
            throw new BusinessException("向量化失败: " + e.getMessage());
        }
    }
    
    // 向量检索
    public List<VectorStoreResult> vectorSearch(
        List<String> datasetIds,
        String question,
        int maxResults,
        double minScore,
        EmbeddingConfig config
    ) {
        OpenAiEmbeddingModel model = modelFactory.createEmbeddingModel(config);
        
        // 查询
        EmbeddingSearchResult<TextSegment> result = embeddingStore.search(
            EmbeddingSearchRequest.builder()
                .filter(new IsIn("dataset_id", datasetIds))
                .maxResults(maxResults)
                .minScore(minScore)
                .queryEmbedding(Embedding.from(model.embed(question).content().vector()))
                .build()
        );
        
        // 转换结果
        return result.matches().stream()
            .map(this::convertToVectorStoreResult)
            .collect(Collectors.toList());
    }
}

## 8️⃣ HybridSearchDomainService

### 职责

- 混合检索（向量+关键词）
- RRF融合算法
- Rerank重排序

### 核心代码

```java
@Service
public class HybridSearchDomainService {
    
    private static final int RRF_K = 60;
    
    private final EmbeddingDomainService embeddingService;
    private final KeywordSearchDomainService keywordSearchService;
    private final RerankDomainService rerankService;
    private final HyDEDomainService hydeService;
    
    public List<DocumentUnitEntity> hybridSearch(HybridSearchConfig config) {
        // 1. HyDE: 生成假设文档
        String hypotheticalDoc = hydeService.generateHypotheticalDocument(
            config.getQuestion(),
            config.getChatModelConfig()
        );
        
        // 2. 并行执行两种检索
        CompletableFuture<List<VectorStoreResult>> vectorFuture = 
            CompletableFuture.supplyAsync(() -> 
                embeddingService.vectorSearch(
                    config.getDataSetIds(),
                    hypotheticalDoc,
                    config.getMaxResults() * 2,
                    config.getMinScore(),
                    config.getEmbeddingConfig()
                )
            );
        
        CompletableFuture<List<VectorStoreResult>> keywordFuture = 
            CompletableFuture.supplyAsync(() -> 
                keywordSearchService.keywordSearch(
                    config.getDataSetIds(),
                    config.getQuestion(),
                    config.getMaxResults() * 2
                )
            );
        
        // 3. 获取结果
        List<VectorStoreResult> vectorResults = vectorFuture.join();
        List<VectorStoreResult> keywordResults = keywordFuture.join();
        
        // 4. RRF融合
        List<VectorStoreResult> fusedResults = fusionWithRRF(
            vectorResults, 
            keywordResults,
            config.getMaxResults()
        );
        
        // 5. Rerank
        if (config.getEnableRerank()) {
            fusedResults = rerankService.rerank(
                fusedResults, 
                config.getQuestion()
            );
        }
        
        return convertToDocumentUnits(fusedResults);
    }
    
    // RRF融合算法
    private List<VectorStoreResult> fusionWithRRF(
        List<VectorStoreResult> vectorResults,
        List<VectorStoreResult> keywordResults,
        int maxResults
    ) {
        Map<String, Double> rrfScores = new HashMap<>();
        
        // 向量检索打分
        for (int i = 0; i < vectorResults.size(); i++) {
            String id = vectorResults.get(i).getId();
            double score = 1.0 / (RRF_K + i + 1);
            rrfScores.merge(id, score, Double::sum);
        }
        
        // 关键词检索打分
        for (int i = 0; i < keywordResults.size(); i++) {
            String id = keywordResults.get(i).getId();
            double score = 1.0 / (RRF_K + i + 1);
            rrfScores.merge(id, score, Double::sum);
        }
        
        // 排序返回topK
        return rrfScores.entrySet().stream()
            .sorted(Map.Entry.<String, Double>comparingByValue().reversed())
            .limit(maxResults)
            .map(e -> findResultById(e.getKey(), vectorResults, keywordResults))
            .filter(Objects::nonNull)
            .collect(Collectors.toList());
    }
}
```
